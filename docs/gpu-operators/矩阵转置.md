<p class="gpu-operator__title">矩阵转置（shared memory / bank conflict / swizzle）</p>
<div class="gpu-operator__meta">
  <span>CUDA</span><span>P02</span><span>bank conflict</span><span>shared memory</span><span>swizzle</span><span>coalescing</span>
</div>

<p class="gpu-operator__label" id="problem">题目</p>

实现 `out[N][M] = in[M][N]ᵀ`，即 `out[j][i] = in[i][j]`，row-major。

<dl class="gpu-operator__io">
  <dt>输入</dt><dd><code>in</code>: [M, N] float32（测试用 4096×4096）</dd>
  <dt>输出</dt><dd><code>out</code>: [N, M] float32</dd>
  <dt>难点</dt><dd><strong>读和写不可能同时合并</strong>：读连续 → 写就是 stride=M 的散射；反过来也一样。必须用 shared memory 做中转。</dd>
  <dt>目标</dt><dd>逼近&quot;纯拷贝&quot;的带宽（读一遍写一遍）</dd>
</dl>

这是**最经典的 bank conflict 面试题**。真正的考点不是转置本身，而是： 你知不知道 shared memory 分 32 个 bank、会不会算某个访问模式冲突多少路、除了 padding 还会不会 swizzle。

<p class="gpu-operator__label" id="signature">给定签名</p>

```cpp
void transpose(const float* in,   // [M, N]
               float* out,        // [N, M]
               int M, int N);
```

<p class="gpu-operator__label" id="tips">Tips</p>

- 先写朴素版，然后**明确指出**：读 `in[row*N+col]` 是合并的，写 `out[col*M+row]` 每个线程差 M×4 字节，一个 warp 要 32 次独立事务。
- 解法就一句话：**用 shared memory 当"转向台"**——合并地读进 tile，转置着从 tile 里取，再合并地写出去。
- 写回时不是简单地交换 threadIdx，**block 的坐标也要交换**：输出的行块由 blockIdx.x 决定。这一步最容易写错，画个图。
- shared 有 32 个 bank，每个 bank 4 字节宽，bank 号 = (字节地址/4) % 32。`tile[32][32]` 按列访问 → 32 路冲突。
- padding 成 `tile[32][33]` 是一行代码的事；再想想为什么工业界（cutlass）用 XOR swizzle 而不是 padding。

<p class="gpu-operator__label" id="prelude">公共前置</p>

```cpp
#include <cstdio>
#include <cstdlib>
#include <cmath>
#include <cuda_runtime.h>

#define CK(x) do{ cudaError_t e=(x); if(e){ printf("CUDA err %s @%d\n",cudaGetErrorString(e),__LINE__); exit(1);} }while(0)
#define CEIL_DIV(a,b) (((a)+(b)-1)/(b))
#define TILE 32
#define ROWS 8      // v3/v4 用：blockDim=(32,8)，每线程搬 4 个元素
```

<p class="gpu-operator__label" id="implementations">分级实现</p>

=== "v0 朴素直搬"

    读 `in[row*N+col]` 同一 warp 内 col 连续 → **读是合并的**；写 `out[col*M+row]` 同一 warp 内地址相差 M\*4 字节 → **32 次独立的 32B 事务，写带宽掉到 1/8~1/32**。这就是转置唯一的难点：读写不可能同时合并，必须借 shared memory 中转。

    ```cpp
    __global__ void transpose_v0(const float* __restrict__ in, float* __restrict__ out, int M, int N){
        int col = blockIdx.x * blockDim.x + threadIdx.x;   // 沿 N
        int row = blockIdx.y * blockDim.y + threadIdx.y;   // 沿 M
        if (row < M && col < N)
            out[(size_t)col * M + row] = in[(size_t)row * N + col];
    }
    void run_v0(const float* in,float* out,int M,int N){
        dim3 b(32,32), g(CEIL_DIV(N,32), CEIL_DIV(M,32));
        transpose_v0<<<g,b>>>(in,out,M,N);
    }
    ```

=== "v1 shared tile（有 bank conflict）"

    用 32x32 的 shared tile 中转：**读全局合并 → 写 shared → sync → 读 shared（转置着读）→ 写全局也合并**。注意写回时的 block 坐标要交换（`bx` 决定输出的行块）。此版 `tile[tx][ty]` 这一列访问让一个 warp 的 32 个线程全落在**同一个 bank**，32 路冲突。

    ```cpp
    __global__ void transpose_v1(const float* __restrict__ in, float* __restrict__ out, int M, int N){
        __shared__ float tile[TILE][TILE];
        int x = blockIdx.x * TILE + threadIdx.x;           // 输入列
        int y = blockIdx.y * TILE + threadIdx.y;           // 输入行
        if (x < N && y < M) tile[threadIdx.y][threadIdx.x] = in[(size_t)y * N + x];
        __syncthreads();
        x = blockIdx.y * TILE + threadIdx.x;     // 输出列（原来的行）：block 坐标交换
        y = blockIdx.x * TILE + threadIdx.y;     // 输出行（原来的列）
        if (x < M && y < N) out[(size_t)y * M + x] = tile[threadIdx.x][threadIdx.y];
    }
    void run_v1(const float* in,float* out,int M,int N){
        dim3 b(TILE,TILE), g(CEIL_DIV(N,TILE), CEIL_DIV(M,TILE));
        transpose_v1<<<g,b>>>(in,out,M,N);
    }
    ```

=== "v2 padding 消 bank conflict"

    shared memory 有 32 个 bank，bank = (地址/4) % 32。`tile[32][32]` 时 `tile[tx][ty]` 的地址是 `tx*32+ty`，tx 变化时 bank 恒为 ty → 32 路冲突。**把行长 padding 成 33**，地址变 `tx*33+ty`，bank = (33tx+ty)%32 = (tx+ty)%32，随 tx 全变 → 无冲突。代价：多 32 个 float 的 shared。

    ```cpp
    __global__ void transpose_v2(const float* __restrict__ in, float* __restrict__ out, int M, int N){
        __shared__ float tile[TILE][TILE + 1];   // ← 只改了这一处
        int x = blockIdx.x * TILE + threadIdx.x;
        int y = blockIdx.y * TILE + threadIdx.y;
        if (x < N && y < M) tile[threadIdx.y][threadIdx.x] = in[(size_t)y * N + x];
        __syncthreads();
        x = blockIdx.y * TILE + threadIdx.x;
        y = blockIdx.x * TILE + threadIdx.y;
        if (x < M && y < N) out[(size_t)y * M + x] = tile[threadIdx.x][threadIdx.y];
    }
    void run_v2(const float* in,float* out,int M,int N){
        dim3 b(TILE,TILE), g(CEIL_DIV(N,TILE), CEIL_DIV(M,TILE));
        transpose_v2<<<g,b>>>(in,out,M,N);
    }
    ```

=== "v3 每线程搬 4 个（32x8）"

    1024 线程/block 会限制 occupancy，且每线程只干一次活。改成 `blockDim=(32,8)`、每线程沿行方向跨 8 行搬 4 个元素：**block 变小、每线程 ILP 变高、索引计算摊薄**，这是 NVIDIA 官方 transpose 样例的经典形态，通常能吃满 ~80% 带宽。

    ```cpp
    __global__ void transpose_v3(const float* __restrict__ in, float* __restrict__ out, int M, int N){
        __shared__ float tile[TILE][TILE + 1];
        int x = blockIdx.x * TILE + threadIdx.x;
        int y = blockIdx.y * TILE + threadIdx.y;
        for (int j = 0; j < TILE; j += ROWS)
            if (x < N && y + j < M) tile[threadIdx.y + j][threadIdx.x] = in[(size_t)(y + j) * N + x];
        __syncthreads();
        x = blockIdx.y * TILE + threadIdx.x;
        y = blockIdx.x * TILE + threadIdx.y;
        for (int j = 0; j < TILE; j += ROWS)
            if (x < M && y + j < N) out[(size_t)(y + j) * M + x] = tile[threadIdx.x][threadIdx.y + j];
    }
    void run_v3(const float* in,float* out,int M,int N){
        dim3 b(TILE,ROWS), g(CEIL_DIV(N,TILE), CEIL_DIV(M,TILE));
        transpose_v3<<<g,b>>>(in,out,M,N);
    }
    ```

=== "v4 XOR swizzle（不用 padding）"

    padding 会浪费 shared、也破坏 128B 对齐（后面写 GEMM/TensorCore 时 `ldmatrix`、`cp.async` 要求对齐，就不能 padding）。工业界做法是 **swizzle**：把元素 (r,c) 存到 `tile[r][c ^ r]`。写入时行固定、列是 c 的置换 → 无冲突；转置读时行随 tid 变、列 `ty^tx` 也全不同 → 同样无冲突。**shared 用量不变**。

    ```cpp
    __global__ void transpose_v4(const float* __restrict__ in, float* __restrict__ out, int M, int N){
        __shared__ float tile[TILE][TILE];                 // 无 padding
        int x = blockIdx.x * TILE + threadIdx.x;
        int y = blockIdx.y * TILE + threadIdx.y;
        if (x < N && y < M) tile[threadIdx.y][threadIdx.x ^ threadIdx.y] = in[(size_t)y * N + x];
        __syncthreads();
        x = blockIdx.y * TILE + threadIdx.x;
        y = blockIdx.x * TILE + threadIdx.y;
        if (x < M && y < N) out[(size_t)y * M + x] = tile[threadIdx.x][threadIdx.y ^ threadIdx.x];
    }
    void run_v4(const float* in,float* out,int M,int N){
        dim3 b(TILE,TILE), g(CEIL_DIV(N,TILE), CEIL_DIV(M,TILE));
        transpose_v4<<<g,b>>>(in,out,M,N);
    }
    ```
