<p class="gpu-operator__title">量化 / 低精度（per-channel、dp4a、fp8 KV cache）</p>
<div class="gpu-operator__meta">

<p class="gpu-operator__label" id="problem">题目</p>

实现对称量化与反量化，并覆盖三种真实推理场景。

`q = round(x / s)`，`x' = q · s`，`s = absmax / 127`（int8）

<dl class="gpu-operator__io">
  <dt>要写的</dt><dd>① per-tensor 量化 ② per-channel（每行一个 scale）③ <code>__dp4a</code> int8 点积
④ fp8 (E4M3) 的 KV cache 量化</dd>
  <dt>要讲的</dt><dd>粒度怎么选、outlier 怎么办、int8 vs fp8、KV cache 为什么用 per-token scale</dd>
  <dt>陷阱</dt><dd>用截断而不是 round-to-nearest、忘记 clamp、整行为 0 时除零</dd>
</dl>

量化是**推理优化岗最实用的一块**：它同时压缩显存和带宽，而 decode 恰恰是 memory-bound 的。 代码不难，但**"为什么这样量化"的问题能问很深**。

<p class="gpu-operator__label" id="signature">给定签名</p>

```cpp
void quantize_per_channel(const float* x,  // [M, N]
                          int8_t* q,       // [M, N]
                          float* scale,    // [M]
                          int M, int N);
```

<p class="gpu-operator__label" id="tips">Tips</p>

- absmax 就是一次 reduce —— 直接复用 P04 的 `blockReduce` 模板。
- ⚠ 用 `__float2int_rn`（round to nearest even），**不要用 `(int)` 强转**——后者是截断，会引入系统性偏差。
- ⚠ 必须 clamp 到 [-127, 127]：浮点舍入可能给出 128，而 int8 存不下（会变成 -128，符号都反了）。
- per-channel：一个 block 负责一行，`blockAllReduce` 求这一行的 absmax。**整行为 0 时要防除零**。
- `__dp4a(a, b, c)`：两个 int 各当 4 个 int8，一条指令 4 次乘加，int32 累加。
- fp8 E4M3 的最大值是 **448**（不是 127），scale 要按它算。

<p class="gpu-operator__label" id="prelude">公共前置</p>

```cpp
#include <cstdio>
#include <cstdlib>
#include <cmath>
#include <cuda_runtime.h>
#include <cuda_fp8.h>

#define CK(x) do{ cudaError_t e=(x); if(e){ printf("CUDA err %s @%d\n",cudaGetErrorString(e),__LINE__); exit(1);} }while(0)
#define CEIL_DIV(a,b) (((a)+(b)-1)/(b))
#define FULL 0xffffffffu
// 对称量化：q = round(x / s)，反量化 x' = q * s，其中 s = absmax / 127（int8）
// 非对称（带 zero-point）：q = round(x/s) + z，多一个偏移，适合分布不对称的激活
__device__ __forceinline__ float warpAllMax(float v){
    #pragma unroll
    for (int o = 16; o > 0; o >>= 1) v = fmaxf(v, __shfl_xor_sync(FULL, v, o));
    return v;
}
__device__ __forceinline__ float blockAllMax(float v){
    __shared__ float sm[32]; __shared__ float r;
    int lane = threadIdx.x & 31, wid = threadIdx.x >> 5, nw = (blockDim.x + 31) >> 5;
    v = warpAllMax(v);
    if (lane == 0) sm[wid] = v;
    __syncthreads();
    if (wid == 0){ float t = (lane < nw) ? sm[lane] : 0.f; t = warpAllMax(t); if (lane == 0) r = t; }
    __syncthreads();
    return r;
}
```

<p class="gpu-operator__label" id="implementations">分级实现</p>

=== "v0 per-tensor 对称量化"

    整个张量共用一个 scale：**先求全局 absmax（就是一次 reduce），再 `q = round(x/s)`**。

    两个必答的细节：① **`__float2int_rn`（round to nearest even）而不是强制类型转换**——后者是截断，会引入系统性偏差；② **必须 clamp 到 [-127, 127]**，否则浮点舍入可能得到 128 而 int8 放不下（溢出成 -128，符号都变了）。

    per-tensor 最省（一个 scale），但 LLM 的激活/权重**动态范围跨通道差异极大**，一个 outlier 通道就能把全局 scale 拉大、把其他通道全压成 0。

    ```cpp
    __global__ void absmax_kernel(const float* __restrict__ x, float* __restrict__ out, size_t n){
        float m = 0.f;
        for (size_t i = blockIdx.x * blockDim.x + threadIdx.x; i < n; i += (size_t)gridDim.x * blockDim.x)
            m = fmaxf(m, fabsf(x[i]));
        m = blockAllMax(m);
        if (threadIdx.x == 0) atomicMax((unsigned*)out, __float_as_uint(m));   // 非负 float 可直接按 uint 比大小
    }
    __global__ void quant_per_tensor(const float* __restrict__ x, int8_t* __restrict__ q,
                                     const float* __restrict__ absmax, size_t n){
        const float s = *absmax / 127.0f;
        const float inv = 1.0f / s;
        for (size_t i = blockIdx.x * blockDim.x + threadIdx.x; i < n; i += (size_t)gridDim.x * blockDim.x){
            int v = __float2int_rn(x[i] * inv);         // round-to-nearest，不是截断
            q[i] = (int8_t)max(-127, min(127, v));                // 必须 clamp
        }
    }
    __global__ void dequant_per_tensor(const int8_t* __restrict__ q, float* __restrict__ y,
                                       const float* __restrict__ absmax, size_t n){
        const float s = *absmax / 127.0f;
        for (size_t i = blockIdx.x * blockDim.x + threadIdx.x; i < n; i += (size_t)gridDim.x * blockDim.x)
            y[i] = (float)q[i] * s;
    }
    ```

=== "v1 per-channel（每行一个 scale）"

    **一个 block 负责一行**：先 blockReduce 求这一行的 absmax，得到该行的 scale，再量化。

    为什么大幅更好：LLM 的权重矩阵里**不同输出通道的幅值能差上百倍**，per-tensor 会让小幅值通道的 int8 只用到几个刻度、精度全丢。per-channel 让每行独占 127 个刻度。

    存储代价只有 M 个 float（相对 M×N 可忽略），而**在 GEMM/GEMV 里 scale 可以提到归约之后再乘一次**（见 P15 v4），几乎零开销。**所以 weight 量化几乎总是 per-channel 起步。**

    ```cpp
    __global__ void quant_per_channel(const float* __restrict__ x, int8_t* __restrict__ q,
                                      float* __restrict__ scale, int N){
        const float* row = x + (size_t)blockIdx.x * N;
        int8_t* qrow = q + (size_t)blockIdx.x * N;
        float m = 0.f;
        for (int i = threadIdx.x; i < N; i += blockDim.x) m = fmaxf(m, fabsf(row[i]));
        m = blockAllMax(m);                                        // 全 block 可见
        const float s = fmaxf(m, 1e-8f) / 127.0f;                  // 防止整行是 0 时除以 0
        if (threadIdx.x == 0) scale[blockIdx.x] = s;
        const float inv = 1.0f / s;
        for (int i = threadIdx.x; i < N; i += blockDim.x){
            int v = __float2int_rn(row[i] * inv);
            qrow[i] = (int8_t)max(-127, min(127, v));
        }
    }
    ```

=== "v2 __dp4a：int8 点积的硬件指令"

    `__dp4a(a, b, c)` 把两个 32 位整数各当成 **4 个 int8**，做 4 组乘加再累加到 c，**一条指令完成 4 次 MAC**，int32 累加不会溢出。

    用法：把 int8 数组按 `int`（4 字节）读，一次处理 4 个元素。相比"逐个 int8 转 float 再乘"，**指令数降到 1/4 且不占浮点单元**。

    这是 **W8A8**（权重和激活都量化）路线的基础。注意最后的反量化：`y = (Σ q_w·q_x) · s_w · s_x`，**两个 scale 都提到最后乘**。

    （Tensor Core 也有 int8 的 mma，吞吐更高；dp4a 是 CUDA Core 上的方案，适合 GEMV 这类没法用 Tensor Core 的场景。）

    ```cpp
    __global__ void dot_dp4a(const int8_t* __restrict__ a, const int8_t* __restrict__ b,
                             int* __restrict__ out, int rows, int K){
        const int row = blockIdx.x, lane = threadIdx.x;
        const int* a4 = reinterpret_cast<const int*>(a + (size_t)row * K);
        const int* b4 = reinterpret_cast<const int*>(b + (size_t)row * K);
        int acc = 0;
        for (int i = lane; i < K / 4; i += blockDim.x)
            acc = __dp4a(a4[i], b4[i], acc);             // 一条指令 = 4 次 int8 乘加
        // int32 归约
        #pragma unroll
        for (int o = 16; o > 0; o >>= 1) acc += __shfl_xor_sync(FULL, acc, o);
        __shared__ int sm[32];
        int wid = lane >> 5, nw = blockDim.x >> 5;
        if ((lane & 31) == 0) sm[wid] = acc;
        __syncthreads();
        if (wid == 0){
            int v = ((lane & 31) < nw) ? sm[lane & 31] : 0;
            #pragma unroll
            for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(FULL, v, o);
            if (lane == 0) out[row] = v;
        }
    }
    ```

=== "v3 fp8 KV cache（per-token scale）"

    KV cache 量化是**长上下文推理的关键优化**：decode 每步都要把整个 cache 读一遍，**fp8 直接把这份带宽和显存砍半**。

    为什么 KV cache 适合用 **fp8 而不是 int8**：fp8（E4M3：4 位指数 3 位尾数）**自带指数，动态范围大得多**，对 attention 里常见的 outlier 更宽容；而 int8 需要精心选 scale。更新 CUDA 架构 还有原生的 fp8 Tensor Core。

    量化粒度选 **per-token（每个 token 的 head\_dim 向量一个 scale）**：新 token 写入 cache 时算一次 scale 就固定了，**历史 cache 完全不用重新量化**——这点非常重要，per-tensor 就做不到。

    ```cpp
    __global__ void quant_kv_fp8(const float* __restrict__ kv, __nv_fp8_storage_t* __restrict__ q,
                                 float* __restrict__ scale, int D){
        // 一个 block 处理一个 token 的一个 head：[D] 个元素共用一个 scale
        const int t = blockIdx.x;
        const float* v = kv + (size_t)t * D;
        float m = 0.f;
        for (int i = threadIdx.x; i < D; i += blockDim.x) m = fmaxf(m, fabsf(v[i]));
        m = blockAllMax(m);
        const float s = fmaxf(m, 1e-8f) / 448.0f;                  // E4M3 的最大可表示值是 448
        if (threadIdx.x == 0) scale[t] = s;
        const float inv = 1.0f / s;
        for (int i = threadIdx.x; i < D; i += blockDim.x)
            q[(size_t)t * D + i] = __nv_cvt_float_to_fp8(v[i] * inv, __NV_SATFINITE, __NV_E4M3);
    }
    __global__ void dequant_kv_fp8(const __nv_fp8_storage_t* __restrict__ q, const float* __restrict__ scale,
                                   float* __restrict__ y, int D){
        const int t = blockIdx.x;
        const float s = scale[t];
        for (int i = threadIdx.x; i < D; i += blockDim.x)
            y[(size_t)t * D + i] = __half2float(__nv_cvt_fp8_to_halfraw(q[(size_t)t * D + i], __NV_E4M3)) * s;
    }
    ```
